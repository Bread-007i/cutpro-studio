   251→  const runSilenceCut = async () => {
   252→    const s = useEditor.getState();
   253→    const clip = s.clips.find((c) => c.id === s.selectedClipId);
   254→    if (!clip || clip.kind !== "media") {
   255→      toast.error("เลือกคลิปที่มีเสียงบนไทม์ไลน์ก่อน");
   256→      return;
   257→    }
   258→    const m = s.media.find((mm) => mm.id === clip.mediaId);
   259→    if (!m?.file || m.type === "image") {
   260→      toast.error("คลิปนี้ไม่มีไฟล์เสียง");
   261→      return;
   262→    }
   263→    setBusy({ active: true, label: "กำลังหาช่วงเสียงเงียบ…", value: 0 });
   264→    try {
   265→      const regions = await detectSilence(m.file, {
   266→        onProgress: (p) => setBusy({ active: true, label: "กำลังหาช่วงเสียงเงียบ…", value: p * 100 }),
   267→      });
   268→      if (regions.length === 0) {
   269→        toast.info("ไม่พบช่วงเสียงเงียบที่ยาวพอ");
   270→        return;
   271→      }
   272→      // build kept segments (inverse of silent regions)
   273→      const srcDur = m.duration;
   274→      const kept: { inPoint: number; duration: number }[] = [];
   275→      let cursor = clip.inPoint;
   276→      const clipEnd = clip.inPoint + clip.duration * clip.props.speed;
   277→      for (const r of regions) {
   278→        const rs = Math.max(clip.inPoint, r.start);
   279→        const re = Math.min(clipEnd, r.end);
   280→        if (re <= rs) continue;
   281→        if (rs > cursor) kept.push({ inPoint: cursor, duration: rs - cursor });
   282→        cursor = Math.max(cursor, re);
   283→      }
   284→      if (cursor < clipEnd) kept.push({ inPoint: cursor, duration: clipEnd - cursor });
   285→      if (kept.length === 0) {
   286→        toast.info("ทั้งคลิปเงียบหมด!");
   287→        return;
   288→      }
   289→      const removed = regions.reduce((a, r) => a + (r.end - r.start), 0);
   290→      s.replaceSegments(clip.id, kept);
   291→      toast.success(`ตัดเสียงเงียบ ${regions.length} ช่วง ประหยัดไป ${removed.toFixed(1)} วินาที`);
   292→    } catch (err) {
   293→      toast.error(err instanceof Error ? err.message : "ตัดเสียงเงียบไม่สำเร็จ");
   294→    } finally {
   295→      setBusy(null);

   347→          { text: ch, fontSize: 40, fontWeight: 600, y: 0.38, textBg: 0.55 },
   348→          t,
   349→          false
   350→        );

   352→        t += seg;
   353→      }
   354→      toast.success(`สร้างซับ ${chunks.length} ชิ้นแล้ว`);
   355→    } catch (err) {
   356→      toast.error(err instanceof Error ? err.message : "สร้างซับไม่สำเร็จ");

   603→function makeThumbFromImage(img: HTMLImageElement): string | undefined {
   604→  try {
   605→    const c = document.createElement("canvas");
   606→    const scale = Math.min(1, 112 / Math.max(img.naturalWidth, img.naturalHeight));
   607→    c.width = Math.max(1, Math.round(img.naturalWidth * scale));
   608→    c.height = Math.max(1, Math.round(img.naturalHeight * scale));
   609→    c.getContext("2d")!.drawImage(img, 0, 0, c.width, c.height);
   610→    return c.toDataURL("image/jpeg", 0.7);
   611→  } catch {
   612→    return undefined;
   613→  }
   614→}

   616→async function thumbFromVideo(v: HTMLVideoElement): Promise<string | undefined> {
   617→  return new Promise((resolve) => {
   618→    const done = () => {
   619→      try {
   620→        const c = document.createElement("canvas");
   621→        const scale = Math.min(1, 112 / Math.max(v.videoWidth || 1, v.videoHeight || 1));
   622→        c.width = Math.max(1, Math.round((v.videoWidth || 112) * scale));
   623→        c.height = Math.max(1, Math.round((v.videoHeight || 63) * scale));
   624→        c.getContext("2d")!.drawImage(v, 0, 0, c.width, c.height);
   625→        resolve(c.toDataURL("image/jpeg", 0.7));
   626→      } catch {
   627→        resolve(undefined);
   628→      }
   629→    };
   630→    if (v.readyState >= 2) {
   631→      done();
   632→    } else {
   633→      v.currentTime = Math.min(1, (v.duration || 2) * 0.15);
   634→      v.onseeked = done;
   635→      setTimeout(done, 2500);
   636→    }
   637→  });
   638→}

   640→function encodeWavLocal(buffer: AudioBuffer): ArrayBuffer {
   641→  const numCh = buffer.numberOfChannels;
   642→  const sr = buffer.sampleRate;
   643→  const len = buffer.length;
   644→  const bytes = len * numCh * 2;
   645→  const ab = new ArrayBuffer(44 + bytes);
   646→  const view = new DataView(ab);
   647→  const writeStr = (off: number, s: string) => {
   648→    for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i));
   649→  };
   650→  writeStr(0, "RIFF");
   651→  view.setUint32(4, 36 + bytes, true);
   652→  writeStr(8, "WAVE");
   653→  writeStr(12, "fmt ");
   654→  view.setUint32(16, 16, true);
   655→  view.setUint16(20, 1, true);
   656→  view.setUint16(22, numCh, true);
   657→  view.setUint32(24, sr, true);
   658→  view.setUint32(28, sr * numCh * 2, true);
   659→  view.setUint16(32, numCh * 2, true);
   660→  view.setUint16(34, 16, true);
   661→  writeStr(36, "data");
   662→  view.setUint32(40, bytes, true);
   663→  const chans: Float32Array[] = [];
   664→  for (let c = 0; c < numCh; c++) chans.push(buffer.getChannelData(c));
   665→  let off = 44;
   666→  for (let i = 0; i < len; i++) {
   667→    for (let c = 0; c < numCh; c++) {
   668→      const s = Math.max(-1, Math.min(1, chans[c][i]));
   669→      view.setInt16(off, s < 0 ? s * 0x8000 : s * 0x7fff, true);
   670→      off += 2;
   671→    }
   672→  }
   673→  return ab;
   674→}
